{{/* A page's category: its `category:` front matter matched case-insensitively against params.categories.list, returned in the configured spelling. Returns "" when categories are disabled, the page has none, or its value is not on the list (warned: a typo should not silently drop a page from its facet). The only reader of the `category:` key — list-json.html and pagefind-keys.html both go through here, so the browse lists and search cannot disagree about a page's category, and the enabled toggle is honoured in one place. Expects a page. */}} {{ $out := "" }} {{ $cfg := site.Params.categories }} {{ with and $cfg $cfg.enabled .Params.category }} {{ $raw := . }} {{ if reflect.IsSlice $raw }} {{ warnidf "category-multi" "%s: category must be a single value, got %v — ignored" $.Path $raw }} {{ else }} {{ range $cfg.list }} {{ if and (not $out) (eq (lower .) (lower (string $raw))) }}{{ $out = . }}{{ end }} {{ end }} {{ if not $out }} {{ warnidf "category-unknown" "%s: category %q is not in params.categories.list — ignored" $.Path (string $raw) }} {{ end }} {{ end }} {{ end }} {{ return $out }}
