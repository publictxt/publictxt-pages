(lower (string $raw))) }}{{ $out = . }}{{ end }} {{ end }} {{ if not $out }} {{ warnidf "category-unknown" "%s: category %q is not in params.categories.list — ignored" $.Path (string $raw) }} {{ end }} {{ end }} {{ end }} {{ return $out }}
