{{/* A page's date, with its provenance. Expects a page in `.page`, and `.detail` true on single pages (where the extra precision is worth the room). `date_source` is written by scripts/dates.py and names the rung the date came from. `front-matter` and `path` are dates an author chose; every other rung is a modification time, so it is labelled "Updated" — an inferred date must never read as a publication date. */}} {{ $p := .page }} {{ $authored := in (slice "front-matter" "path") ($p.Params.date_source | default "") }} {{ if not $p.Date.IsZero }} {{ if $authored }} {{ $p.Date.Format "2 Jan 2006" }} {{ if and .detail (gt (int (sub $p.Lastmod.Unix $p.Date.Unix)) 86400) }} · updated {{ $p.Lastmod.Format "2 Jan 2006" }} {{ end }} {{ else }} Updated {{ $p.Date.Format "2 Jan 2006" }} {{ end }} {{ end }}
