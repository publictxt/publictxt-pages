{{/* The URL for a Markdown destination `path` written in `page`. Returns it. A relative path is a file in one of two places, found as a Hugo resource and linked by its URL (folder names aren't URLs — slugs): beside the note (a resource of its folder's section, subfolders included), or in a top-level folder of files, e.g. `../../media/x.mp3` (a home resource — such folders get no `_index.md`). Elsewhere: left as written, with a warning. Absolute paths and URLs: as written. */}} {{- $u := urls.Parse .path -}} {{- $url := .path -}} {{- if and $u.Path (not $u.Host) (not $u.Scheme) (not (hasPrefix $u.Path "/")) -}} {{- $p := .page -}} {{- $r := $p.CurrentSection.Resources.Get $u.Path -}} {{- with $p.File -}} {{- $f := path.Join (replace .Dir "\\" "/") $u.Path -}} {{- if and (not $r) (eq (len (split $f "/")) 2) }}{{ $r = site.Home.Resources.Get $f }}{{ end -}} {{- end -}} {{- with $r }}{{ $url = .RelPermalink }}{{ else }}{{ warnf "%s: file not beside the note or in a top-level folder: %s" (or (and $p.File $p.File.Path) $p.RelPermalink) $.path }}{{ end -}} {{- end -}} {{- return $url -}}
