{{/* Pager for a paginated list. Expects a dict: paginator — a Paginator order — the list's order string (ordered.html); date orders label the steps Newer/Older, title orders Previous/Next. Renders nothing for a single page. Shows first/last, a window around the current page, and the two steps — at most ~9 links however long the list. */}} {{ $pg := .paginator }} {{ if gt $pg.TotalPages 1 }} {{ $cur := $pg.PageNumber }}{{ $total := $pg.TotalPages }} {{ $field := index (split (trim (lower (.order | default "updated")) " ") " ") 0 }} {{ $back := "‹ Previous" }}{{ $fwd := "Next ›" }} {{ if ne $field "title" }} {{ $asc := strings.HasSuffix (lower .order) " asc" }} {{ $back = cond $asc "‹ Older" "‹ Newer" }}{{ $fwd = cond $asc "Newer ›" "Older ›" }} {{ end }}
{{ if $pg.HasPrev }}{{ $back }}{{ else }}{{ $back }}{{ end }}
    {{ $prev := 0 }} {{ range $pg.Pagers }} {{ $n := .PageNumber }} {{ if or (le $n 2) (ge $n (sub $total 1)) (and (ge $n (sub $cur 2)) (le $n (add $cur 2))) }} {{ if gt (sub $n $prev) 1 }}
  1. …
    2. {{ end }}
  2. {{ if eq $n $cur }}{{ $n }}{{ else }}{{ $n }}{{ end }}
    3. {{ $prev = $n }} {{ end }} {{ end }}
{{ if $pg.HasNext }}{{ $fwd }}{{ else }}{{ $fwd }}{{ end }}
{{ end }}
