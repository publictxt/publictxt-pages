{{/* Order a page collection. Expects a dict: pages — a page collection order — "[ ]", where field is one of updated | created | title and direction is asc | desc. Defaults: dates newest first, titles A→Z. Anything else falls back to "updated" — the site's one meaning of recency (recent.html), which stays the canonical definition for that case. */}} {{ $parts := split (trim (lower (.order | default "updated")) " ") " " }} {{ $field := index $parts 0 }} {{ $dir := "" }}{{ if gt (len $parts) 1 }}{{ $dir = index $parts 1 }}{{ end }} {{ $pages := .pages }} {{ if eq $field "created" }} {{ $pages = sort $pages "Date" (cond (eq $dir "asc") "asc" "desc") }} {{ else if eq $field "title" }} {{ $pages = sort $pages "Title" (cond (eq $dir "desc") "desc" "asc") }} {{ else if eq $dir "asc" }} {{ $pages = sort $pages "Lastmod" "asc" }} {{ else }} {{ $pages = partial "recent.html" $pages }} {{ end }} {{ return $pages }}
