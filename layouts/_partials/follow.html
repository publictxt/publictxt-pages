{{/* A follow — a page about a repo this one follows — as sync marked it (follow_name:, follow_url:): a dict of name (what its forks call their source), url (the repo; "" for a local follow), label (the URL without its scheme; "local repository"), rating (this author's of the repo, int 1–5, else 0), description (`description:`) and forks (how many pages came through it). False for any other page. A follow's `rating:` is the repo's, not the page's: read here, and rating.html gives 0. Expects a page. */}} {{ $out := false }} {{ with .Params.follow_name }} {{ $url := default "" $.Params.follow_url }} {{ $rating := 0 }} {{ with $.Params.rating }}{{ if findRE `^[1-5]$` (printf "%v" .) }}{{ $rating = int (printf "%v" .) }}{{ end }}{{ end }} {{ $forks := 0 }} {{ with $.File }}{{ $forks = len (where site.RegularPages "Params.source_follow" .Path) }}{{ end }} {{ $out = dict "name" (printf "%v" .) "url" $url "label" (cond (ne $url "") (strings.TrimPrefix "https://" $url) "local repository") "rating" $rating "description" (default "" $.Params.description) "forks" $forks }} {{ end }} {{ return $out }}
